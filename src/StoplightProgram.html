Stoplight with Sound Program

Elegoo Uno R3 + LED's + MP3 Module + Aux Output


#include <SoftwareSerial.h>

SoftwareSerial mp3Serial(10, 11);
const int busyPin = 12;
const int LED1 = 2; // red
const int LED2 = 3; // yellow
const int LED3 = 4; // green
const int LED4 = 5; // red
const int LED5 = 6; // yellow
const int LED6 = 7; // green

void setup() {
  Serial.begin(9600);
  mp3Serial.begin(9600);
  pinMode(busyPin, INPUT);
  pinMode(LED1, OUTPUT);
  pinMode(LED2, OUTPUT);
  pinMode(LED3, OUTPUT);
  pinMode(LED4, OUTPUT);
  pinMode(LED5, OUTPUT);
  pinMode(LED6, OUTPUT);
  delay(2000);
  Serial.println("Starting LED/audio sequence...");
}

void loop() {
  digitalWrite(LED1, LOW);  // red
  digitalWrite(LED2, LOW);  // yellow
  digitalWrite(LED3, HIGH); // green
  digitalWrite(LED4, HIGH); // red
  digitalWrite(LED5, LOW);  // yellow
  digitalWrite(LED6, LOW);  // green

  if (random(100) < 30) {
    playSong(1);
    waitUntilDone();
  } else {
    delay(8000);
  }

  delay(1000);  // always runs

  digitalWrite(LED1, LOW);  
  digitalWrite(LED2, HIGH); 
  digitalWrite(LED3, LOW);  
  digitalWrite(LED4, HIGH); 
  digitalWrite(LED5, LOW);  
  digitalWrite(LED6, LOW);  
  delay(4000);

  digitalWrite(LED1, HIGH); 
  digitalWrite(LED2, LOW);  
  digitalWrite(LED3, LOW);  
  digitalWrite(LED4, HIGH); 
  digitalWrite(LED5, LOW);  
  digitalWrite(LED6, LOW);  
  delay(4000);

  digitalWrite(LED1, HIGH); 
  digitalWrite(LED2, LOW);  
  digitalWrite(LED3, LOW);  
  digitalWrite(LED4, LOW);  
  digitalWrite(LED5, LOW);  
  digitalWrite(LED6, HIGH); 

  if (random(100) < 30) {
    playSong(2);
    waitUntilDone();
  } else {
    delay(8000);
  }

  delay(1000);

  digitalWrite(LED1, HIGH); 
  digitalWrite(LED2, LOW);  
  digitalWrite(LED3, LOW);  
  digitalWrite(LED4, LOW);  
  digitalWrite(LED5, HIGH); 
  digitalWrite(LED6, LOW);  
  delay(4000);

  digitalWrite(LED1, HIGH); 
  digitalWrite(LED2, LOW);  
  digitalWrite(LED3, LOW);  
  digitalWrite(LED4, HIGH); 
  digitalWrite(LED5, LOW);  
  digitalWrite(LED6, LOW);  
  delay(4000);
}

void playSong(int songNumber) {
  Serial.print("Playing track: ");
  Serial.println(songNumber);
  mp3Serial.write((uint8_t)0xAA);
  mp3Serial.write((uint8_t)0x07);
  mp3Serial.write((uint8_t)0x02);
  mp3Serial.write((uint8_t)0x00);
  mp3Serial.write((uint8_t)songNumber);
  mp3Serial.write((uint8_t)((0xAA + 0x07 + 0x02 + 0x00 + songNumber) & 0xFF));
  delay(500);
}

void waitUntilDone() {
  while (digitalRead(busyPin) == LOW) {
  }
}
